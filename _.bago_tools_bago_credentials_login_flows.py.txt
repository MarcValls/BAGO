0: """LoginFlowsMixin — flujos interactivos de /login para CredentialManager.
1: 
2: Separado de manager.py para mantener la lógica de UI aislada del estado.
3: Se mezcla con CredentialManager mediante herencia múltiple.
4: """
5: 
6: import subprocess
7: 
8: from prompt_toolkit import prompt as pt_prompt
9: 
10: from ..ui import console
11: from .accounts import AccountManager
12: from .flows import (
13:     flow_github,
14:     flow_openai,
15:     flow_ollama_cloud,
16:     flow_ollama_service,
17:     flow_opencode,
18:     flow_gittoken,
19:     flow_api_key,
20:     flow_huggingface,
21:     flow_sendcm,
22: )
23: 
24: 
25: class LoginFlowsMixin:
26:     """Métodos interactivos de /login. Requiere que self tenga:
27:     - self._accounts (AccountManager)
28:     - self._creds (dict)
29:     - self._save()
30:     - self.set(provider, key)
31:     - self.PROVIDERS / self.ALIASES
32:     """
33: 
34:     def do_login(self, alias: str) -> str:
35:         """Registra/gestiona credenciales.
36: 
37:         Subcomandos multi-cuenta:
38:           list                           — lista todas las cuentas
39:           switch <account-id>            — activa una cuenta
40:           remove <account-id>            — elimina una cuenta
41:           add <provider> [nombre]        — agrega una cuenta nueva
42:           <provider>                     — flujo clásico (reemplaza la 1ª cuenta)
43:         """
44:         parts = alias.strip().split(None, 2)
45:         sub = parts[0].lower() if parts else ""
46: 
47:         if sub == "list":
48:             return self._login_list()
49:         if sub == "switch":
50:             return self._login_switch(parts)
51:         if sub == "remove":
52:             return self._login_remove(parts)
53:         if sub == "add":
54:             return self._login_add(parts)
55: 
56:         # ── Flujo clásico ────────────────────────────────────────────────────
57:         name = self.ALIASES.get(alias.lower(), alias.lower())
58:         info = self.PROVIDERS.get(name)
59:         if not info:
60:             return (
61:                 f"Provider '{alias}' desconocido.\n"
62:                 f"  Providers: {', '.join(self.PROVIDERS)}\n"
63:                 f"  Subcomandos: add · list · switch · remove"
64:             )
65:         return self._login_classic(name, info)
66: 
67:     # ── Subcomandos multi-cuenta ─────────────────────────────────────────────
68: 
69:     def _login_list(self) -> str:
70:         lines = self._accounts.summary_lines()
71:         console.print("\n[bold]Cuentas registradas:[/bold]")
72:         for line in lines:
73:             console.print(line)
74:         console.print(
75:             "\n[dim]  /login add <provider> [nombre]  — agregar cuenta nueva"
76:             "\n  /login switch <id>               — activar cuenta"
77:             "\n  /login remove <id>               — eliminar cuenta[/dim]"
78:         )
79:         return ""
80: 
81:     def _login_switch(self, parts: list) -> str:
82:         if len(parts) < 2:
83:             return "[red]Uso: /login switch <account-id>  (ej: github-2)[/red]"
84:         account_id = parts[1]
85:         if self._accounts.set_active(account_id):
86:             self._accounts.apply_active_credentials()
87:             acc = self._accounts.find(account_id)
88:             label = acc.get("label", account_id) if acc else account_id
89:             return f"[green]✓ Cuenta activa: {account_id} — {label}[/green]"
90:         return (
91:             f"[red]Cuenta '{account_id}' no encontrada. "
92:             f"Usa /login list para ver las disponibles.[/red]"
93:         )
94: 
95:     def _login_remove(self, parts: list) -> str:
96:         if len(parts) < 2:
97:             return "[red]Uso: /login remove <account-id>  (ej: github-2)[/red]"
98:         account_id = parts[1]
99:         if self._accounts.remove(account_id):
100:             self._accounts.apply_active_credentials()
101:             return f"[green]✓ Cuenta '{account_id}' eliminada.[/green]"
102:         return f"[red]Cuenta '{account_id}' no encontrada.[/red]"
103: 
104:     def _login_add(self, parts: list) -> str:
105:         if len(parts) < 2:
106:             return (
107:                 "[red]Uso: /login add <provider> [nombre][/red]\n"
108:                 "[dim]  Providers: github, openai, anthropic, openrouter, gemini, ollama_cloud[/dim]"
109:             )
110:         provider_raw = parts[1].lower()
111:         custom_label = parts[2] if len(parts) > 2 else ""
112:         provider = self.ALIASES.get(provider_raw, provider_raw)
113:         am_provider = self._bago_to_am_provider(provider)
114:         if am_provider not in AccountManager.PROVIDER_ENV and am_provider != "ollama":
115:             return f"[red]Provider '{provider_raw}' no soportado para multi-cuenta.[/red]"
116:         return self._do_add_account(am_provider, custom_label)
117: 
118:     # ── Flujo clásico por tipo de login ─────────────────────────────────────
119: 
120:     def _login_classic(self, name: str, info: dict) -> str:
121:         ltype = info["login_type"]
122:         if ltype == "github":
123:             return self._flow_github()
124:         if ltype == "gh_cli":          # alias legacy
125:             return self._flow_github()
126:         if ltype == "openai_cli":
127:             return self._flow_openai()
128:         if ltype == "api_key":
129:             return self._flow_api_key(name, info)
130:         if ltype == "ollama_cloud":
131:             return self._flow_ollama_cloud()
132:         if ltype == "opencode_cli":
133:             return self._flow_opencode()
134:         if ltype == "service":
135:             return self._flow_ollama_service()
136:         if ltype == "sendcm":
137:             return self._flow_sendcm()
138:         if ltype == "gitlab":
139:             return self._flow_gittoken("gitlab", "GitLab",
140:                                         "https://gitlab.com/api/v4/user",
141:                                         "PRIVATE-TOKEN")
142:         if ltype == "codeberg":
143:             return self._flow_gittoken("codeberg", "Codeberg",
144:                                         "https://codeberg.org/api/v1/user",
145:                                         "Authorization", prefix="token ")
146:         if ltype == "huggingface":
147:             return self._flow_huggingface()
148:         return f"[red]Tipo de login '{ltype}' no reconocido.[/red]"
149: 
150:     def _flow_github(self) -> str:
151:         return flow_github(self)
152: 
153:     def _flow_openai(self) -> str:
154:         return flow_openai(self)
155: 
156:     def _flow_api_key(self, name: str, info: dict) -> str:
157:         return flow_api_key(self, name, info)
158: 
159:     def _flow_ollama_cloud(self) -> str:
160:         return flow_ollama_cloud(self)
161: 
162:     def _flow_opencode(self) -> str:
163:         return flow_opencode(self)
164: 
165:     def _flow_ollama_service(self) -> str:
166:         return flow_ollama_service(self)
167: 
168:     def _flow_gittoken(self, provider: str, label: str,
169:                        verify_url: str, auth_header: str,
170:                        prefix: str = "") -> str:
171:         return flow_gittoken(self, provider, label, verify_url, auth_header, prefix)
172: 
173:     def _flow_huggingface(self) -> str:
174:         return flow_huggingface(self)
175: 
176:     def _flow_sendcm(self) -> str:
177:         return flow_sendcm(self)
178: 
179:     # ── Wizard de nueva cuenta ───────────────────────────────────────────────
180: 
181:     def _do_add_account(self, provider: str, custom_label: str = "") -> str:
182:         """Flujo interactivo para agregar una cuenta nueva de cualquier provider."""
183:         am = self._accounts
184:         existing = am.accounts_for(provider)
185:         n = len(existing)
186:         plabel = AccountManager.PROVIDER_LABELS.get(provider, provider)
187: 
188:         console.print(
189:             f"\n[bold]Agregar cuenta nueva — {plabel}[/bold]"
190:             + (f"\n[dim]  Ya tienes {n} cuenta{'s' if n != 1 else ''} de este tipo.[/dim]"
191:                if n > 0 else "")
192:         )
193: 
194:         if not custom_label:
195:             default = f"{plabel} #{n + 1}"
196:             raw = pt_prompt(f"Nombre/etiqueta [{default}]: ").strip()
197:             label = raw or default
198:         else:
199:             label = custom_label
200: 
201:         if provider == "github":
202:             console.print("[dim]Ejecutando gh auth login...[/dim]")
203:             result = subprocess.run(["gh", "auth", "login"])
204:             if result.returncode != 0:
205:                 return "[red]gh auth login fallido.[/red]"
206:             try:
207:                 token = subprocess.check_output(["gh", "auth", "token"], text=True).strip()
208:                 account_id = am.add("github", label, token, "token", make_active=True)
209:                 am.apply_active_credentials()
210:                 return f"[green]✓ Cuenta añadida: {account_id} — {label}  ({token[:4]}…{token[-4:]})[/green]"
211:             except Exception as e:
212:                 return f"[red]Error obteniendo token: {e}[/red]"
213: 
214:         if provider == "openai":
215:             console.print(
216:                 "[bold]OpenAI / GPT — elige método:[/bold]\n"
217:                 "  [yellow]1[/yellow]  codex login  (GPT Plus — abre navegador)\n"
218:                 "  [yellow]2[/yellow]  API key      (platform.openai.com)\n"
219:             )
220:             choice = pt_prompt("Opción [1/2]: ").strip()
221:             if choice == "1":
222:                 result = subprocess.run(["codex", "login"])
223:                 if result.returncode == 0:
224:                     account_id = am.add("openai", label, "__codex_oauth__", "oauth", make_active=True)
225:                     return f"[green]✓ Cuenta añadida: {account_id} — {label} (codex OAuth)[/green]"
226:                 return "[red]codex login fallido.[/red]"
227:             else:
228:                 key = pt_prompt("OpenAI API Key: ", is_password=True).strip()
229:                 if not key:
230:                     return "Cancelado."
231:                 account_id = am.add("openai", label, key, "api_key", make_active=True)
232:                 am.apply_active_credentials()
233:                 return f"[green]✓ Cuenta añadida: {account_id} — {label}[/green]"
234: 
235:         # Genérico: cualquier provider con API key
236:         url = self.PROVIDERS.get(provider, {}).get("url", "")
237:         if url:
238:             console.print(f"[dim]Obtén tu clave en: {url}[/dim]")
239:         key = pt_prompt(f"API Key para {label}: ", is_password=True).strip()
240:         if not key:
241:             return "Cancelado."
242:         account_id = am.add(provider, label, key, "api_key", make_active=True)
243:         am.apply_active_credentials()
244:         return f"[green]✓ Cuenta añadida: {account_id} — {label}[/green]"
245: 
246:     # ── Helpers ─────────────────────────────────────────────────────────────
247: 
248:     def _bago_to_am_provider(self, name: str) -> str:
249:         """Convierte nombre de CredentialManager a tipo de AccountManager."""
250:         return {
251:             "github":       "github",
252:             "openai":       "openai",
253:             "anthropic":    "anthropic",
254:             "openrouter":   "openrouter",
255:             "gemini":       "gemini",
256:             "ollama_cloud": "ollama_cloud",
257:         }.get(name, name)
